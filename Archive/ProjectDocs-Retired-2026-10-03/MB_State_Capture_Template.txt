-----------------------------------------------------------------
COMMAND 1: SecurityCenter2 query
-----------------------------------------------------------------
Get-CimInstance -Namespace root\SecurityCenter2 -ClassName AntiVirusProduct | Select displayName, productState | Format-List

OUTPUT:
[paste here]

-----------------------------------------------------------------
COMMAND 2: Defender status
-----------------------------------------------------------------
Get-MpComputerStatus | Select RealTimeProtectionEnabled, AntivirusEnabled, AMRunningMode, AMProductVersion | Format-List

OUTPUT:
[paste here]

-----------------------------------------------------------------
COMMAND 3: MB registry key check
-----------------------------------------------------------------
Get-ChildItem "HKLM:\SOFTWARE" | Where-Object { $_.Name -like "*Malwarebytes*" } | Select Name

OUTPUT:
[paste here]

-----------------------------------------------------------------
COMMAND 4: MB registry key (alternate path)
-----------------------------------------------------------------
Get-ChildItem "HKLM:\SOFTWARE\WOW6432Node" | Where-Object { $_.Name -like "*Malwarebytes*" } | Select Name

OUTPUT:
[paste here]

-----------------------------------------------------------------
COMMAND 5: MB service state
-----------------------------------------------------------------
Get-Service | Where-Object { $_.Name -like "*Malwarebytes*" -or $_.Name -like "*MBAMService*" } | Select Name, Status, StartType

OUTPUT:
[paste here]

-----------------------------------------------------------------
COMMAND 6: Installed programs check
-----------------------------------------------------------------
Get-ItemProperty "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*" | Where-Object { $_.DisplayName -like "*Malwarebytes*" } | Select DisplayName, DisplayVersion

[paste here]
