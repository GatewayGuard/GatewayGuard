Get-ChildItem "$env:LOCALAPPDATA\Microsoft\Edge\User Data" -Directory |
    Where-Object { Test-Path (Join-Path $_.FullName 'Preferences') } |
    ForEach-Object {
      $p = Join-Path $_.FullName 'Preferences'
      $j = Get-Content $p -Raw | ConvertFrom-Json
      [PSCustomObject]@{
        Profile                   = $_.Name
        credentials_enable_service= $j.credentials_enable_service
        startup_boost_enabled     = $j.startup_boost_enabled
        background_mode_enabled   = $j.background_mode.enabled
      }
    } | Format-Table -AutoSize
